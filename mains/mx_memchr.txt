int main()
{
// Исходный массив
   unsigned char src[15]= "1234567890";
   // Переменная, в которую будет сохранен указатель
   // на искомый символ.
   char *sym;

   // Поиск требуемого символа
   sym = mx_memchr (src, '4', 10);
   // Вывод исходного массива
   printf ("sym: %s\n",sym);

   return 0;
}
