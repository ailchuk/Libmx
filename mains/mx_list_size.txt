// Func to display node!!

void display(t_list *head)
{
    while (head != NULL)
    {
        printf("%s   ", head->data);
        head = head->next;
    }
    printf("\n");
}

int main()
{
    t_list *list = mx_create_node("Hell");
    list->next = mx_create_node("TOTT");
    list->next->next =mx_create_node("OOOOOOO");
    list->next->next->next = mx_create_node("Aloha-Loha");
    list->next->next->next->next = mx_create_node("good-bad-bod-god-dog");
    display(list);
    printf("%d\n", mx_list_size(list));
}
