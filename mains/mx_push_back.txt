
// Func to display node!!

void display(t_list *head)
{
    while (head != NULL)
    {
        printf("%s   ", head->data);
        head = head->next;
    }
    printf("\n");
}

int main() 
{
    t_list listok = *mx_create_node("HellO");
    display(&listok);    

    
    mx_push_back(&listok.next, "hi");
       
    mx_push_back(&listok.next->next, "keke");

    display(&listok);

}
